create database if not exists todolist;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(200),
    email VARCHAR(200),
    senha VARCHAR(200),
    adm BOOLEAN
);

CREATE TABLE quadros (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT, 
    titulo VARCHAR(300),
    favorito BOOLEAN,
    cor_quadro VARCHAR(100),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE blocos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    quadro_id INT,
    titulo VARCHAR(300),
    cor_bloco VARCHAR(100),
    FOREIGN KEY (quadro_id) REFERENCES quadros(id)
);

CREATE TABLE tarefas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    bloco_id INT,
    titulo TEXT,
    descricao TEXT,
    item VARCHAR(300),
    status VARCHAR(50),
    caminho_anexo TEXT,
    FOREIGN KEY (bloco_id) REFERENCES blocos(id)
);
