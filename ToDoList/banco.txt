create database if not exists todolist;

create table usuarios {
    id      int auto_increment  primary_key,
    nome    varchar(200),
    email   varchar(200),
    senha   varchar(200),
    adm     boolean
};

create table quadros {
    id          int auto_increment  primary_key,
    titulo      varchar(300),
    favorito    boolean,
    cor_quadro  varchar(100),
    foreign_key (usuario_id) references usuarios(id)
};

create table blocos {
    id          int auto_increment  primary_key,
    titulo      varchar(300),
    cor_bloco   varchar(100),
    foreign_key (quadro_id) references quadros(id)
};

create table tarefas {
    id              int auto_increment  primary_key,
    titulo          text,
    descricao       text,
    item            varchar(300),
    status          varchar(50),
    caminho_anexo   text,
    foreign_key(bloco_id) references tarefas(id)        
};
